package com.masterclass.inventory.service; import com.masterclass.inventory.dto.ReportDtos.*; import com.masterclass.inventory.repository.*; import lombok.RequiredArgsConstructor; import org.springframework.stereotype.Service; import org.springframework.transaction.annotation.Transactional; import java.math.BigDecimal; import java.time.*; import java.util.*;
@Service @RequiredArgsConstructor public class ReportService { private final ProductRepository products; private final SaleRepository sales; @Transactional(readOnly=true) public List<ProductInventoryReport> inventory(){return products.findAll().stream().map(this::toReport).toList();} @Transactional(readOnly=true) public List<ProductInventoryReport> lowStock(){return products.findLowStock().stream().map(this::toReport).toList();} @Transactional(readOnly=true) public InventoryValueReport inventoryValue(){ var total=products.totalInventoryValue(); return new InventoryValueReport(total!=null?total:BigDecimal.ZERO, products.count());} private ProductInventoryReport toReport(com.masterclass.inventory.entity.Product p){return new ProductInventoryReport(p.getId(),p.getSku(),p.getName(),p.getCategory().getName(),p.getCurrentStock(),p.getReorderLevel(),p.getUnitPrice().multiply(BigDecimal.valueOf(p.getCurrentStock())));} @Transactional(readOnly=true) public DailySalesReport daily(LocalDate date){var from=date.atStartOfDay().atOffset(ZoneOffset.UTC); var to=date.plusDays(1).atStartOfDay().atOffset(ZoneOffset.UTC); var rows=sales.findSalesBetween(from,to); var total=rows.stream().map(s->s.getTotalAmount()).reduce(BigDecimal.ZERO,BigDecimal::add); return new DailySalesReport(date,rows.size(),total);} }
