package com.masterclass.inventory.service; import com.masterclass.inventory.dto.InventoryDtos.*; import com.masterclass.inventory.entity.*; import com.masterclass.inventory.exception.*; import com.masterclass.inventory.repository.*; import lombok.RequiredArgsConstructor; import org.springframework.stereotype.Service; import org.springframework.transaction.annotation.Transactional; import java.time.OffsetDateTime; import java.util.*;
@Service @RequiredArgsConstructor public class InventoryService { private final ProductRepository products; private final InventoryMovementRepository movements; @Transactional public MovementResponse replenish(InventoryCommand c){return move(c.productId(), MovementType.REPLENISHMENT, c.quantity(), c.user());} @Transactional public MovementResponse adjust(AdjustmentCommand c){ if(c.type()!=MovementType.ADJUSTMENT_IN && c.type()!=MovementType.ADJUSTMENT_OUT) throw new BusinessException("Invalid adjustment type"); return move(c.productId(), c.type(), c.quantity(), c.user()); } @Transactional private MovementResponse move(Long pid, MovementType type, int qty, String user){ Product p=products.findById(pid).orElseThrow(()->new NotFoundException("Product not found")); if(qty<=0) throw new BusinessException("Quantity must be positive"); if(type==MovementType.REPLENISHMENT||type==MovementType.ADJUSTMENT_IN) p.increaseStock(qty); else { if(p.getCurrentStock()<qty) throw new BusinessException("Insufficient stock for product "+p.getSku()); p.decreaseStock(qty);} InventoryMovement m=movements.save(InventoryMovement.builder().product(p).movementType(type).quantity(qty).date(OffsetDateTime.now()).user(user).build()); return new MovementResponse(m.getId(),p.getId(),p.getSku(),m.getMovementType(),m.getQuantity(),m.getDate(),m.getUser()); } @Transactional(readOnly=true) public List<MovementResponse> history(Long productId){return movements.findByProductIdOrderByDateDesc(productId).stream().map(m->new MovementResponse(m.getId(),m.getProduct().getId(),m.getProduct().getSku(),m.getMovementType(),m.getQuantity(),m.getDate(),m.getUser())).toList();} }
